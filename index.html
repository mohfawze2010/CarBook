window.registerApp = function() { var email = document.getElementById('userEmail').value; var pass = document.getElementById('userPassword').value; var msg = document.getElementById('authMessage'); if(!email || !pass) { msg.innerText = 'أدخل الإيميل والرقم السري أولاً'; msg.style.display = 'block'; return; } // إجبار النظام على حفظ تسجيل الدخول محلياً auth.setPersistence(firebase.auth.Auth.Persistence.LOCAL) .then(function() { return auth.createUserWithEmailAndPassword(email, pass); }) .then(function() { msg.style.display = 'none'; alert('✅ تم إنشاء الحساب وتسجيل الدخول بنجاح!'); }).catch(function(error) { msg.innerText = translateAuthError(error); msg.style.display = 'block'; }); };
